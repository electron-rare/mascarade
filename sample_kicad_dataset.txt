# Python script for KiCad automation
def generate_footprint(name, size):
    # Create footprint
    footprint = Footprint(name)
    # Add pads
    for i in range(2):
        pad = Pad(i+1, "thru_hole", shape="rect", at=[i*size, 0], size=[1.5, 1.5], drill=0.8)
        footprint.add(pad)
    return footprint
